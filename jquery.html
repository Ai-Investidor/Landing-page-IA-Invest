$('#para-scene').on('mousemove', function(e) {

  const x = e.clientX - rect.left - rect.width / 2;

#23

Typewriter & Auto-Delete Carousel

Typewriter Engine
const phrases = ["Inovação Ágil", "UX Futurista", "Performance Pura"];
let pIdx = 0, charIdx = 0, deleting = false;

function typeStep() {
  const cur = phrases[pIdx];
  $('#typewriter-el').text(cur.substring(0, charIdx));
  
  if (!deleting && charIdx++ === cur.length) {
    deleting = true; setTimeout(typeStep, 1000); return;
  }
  if (deleting && charIdx-- === 0) {
    deleting = false; pIdx = (pIdx + 1) % phrases.length;
  }
  setTimeout(typeStep, deleting ? 40 : 90);
}
typeStep();
